cmake_minimum_required(VERSION 3.24)
project(ZToolsPathAccess LANGUAGES CXX)
add_library(ZToolsPathAccess STATIC path_access.cpp)
if(APPLE)
  enable_language(OBJCXX)
  target_sources(ZToolsPathAccess PRIVATE trash_mac.mm)
  target_compile_options(ZToolsPathAccess PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
  target_link_libraries(ZToolsPathAccess PUBLIC "-framework Foundation")
endif()
target_compile_features(ZToolsPathAccess PUBLIC cxx_std_17)
target_include_directories(ZToolsPathAccess PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(ZToolsPathAccess PRIVATE /W4 /WX /utf-8 /sdl)
else()
  target_compile_options(ZToolsPathAccess PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(ZToolsPathAccessTest path_access_test.cpp)
target_link_libraries(ZToolsPathAccessTest PRIVATE ZToolsPathAccess)
add_executable(ZToolsPathAccessNamesTest path_names_test.cpp)
target_link_libraries(ZToolsPathAccessNamesTest PRIVATE ZToolsPathAccess)
add_executable(ZToolsPathAccessBroker broker.cpp)
target_link_libraries(ZToolsPathAccessBroker PRIVATE ZToolsPathAccess)
if(MSVC)
  target_compile_options(ZToolsPathAccessTest PRIVATE /W4 /WX /utf-8 /sdl)
  target_compile_options(ZToolsPathAccessNamesTest PRIVATE /W4 /WX /utf-8 /sdl)
  target_compile_options(ZToolsPathAccessBroker PRIVATE /W4 /WX /utf-8 /sdl)
else()
  target_compile_options(ZToolsPathAccessTest PRIVATE -Wall -Wextra -Werror)
  target_compile_options(ZToolsPathAccessNamesTest PRIVATE -Wall -Wextra -Werror)
  target_compile_options(ZToolsPathAccessBroker PRIVATE -Wall -Wextra -Werror)
endif()
