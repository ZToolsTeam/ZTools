cmake_minimum_required(VERSION 3.24)
project(ZToolsMacNativeAbiFixture LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(_zt_runtime_dir "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${_zt_runtime_dir}")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_DEBUG "${_zt_runtime_dir}")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${_zt_runtime_dir}")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELWITHDEBINFO "${_zt_runtime_dir}")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_MINSIZEREL "${_zt_runtime_dir}")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${_zt_runtime_dir}")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_DEBUG "${_zt_runtime_dir}")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_RELEASE "${_zt_runtime_dir}")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_RELWITHDEBINFO "${_zt_runtime_dir}")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_MINSIZEREL "${_zt_runtime_dir}")

add_library(ZToolsMacNativeAbiFixture SHARED fixture/abi_fixture.cpp)
add_library(ZToolsMacNativeUnsupportedAbiFixture SHARED fixture/abi_fixture.cpp)
target_compile_definitions(ZToolsMacNativeUnsupportedAbiFixture PRIVATE ZT_MAC_FIXTURE_ABI_VERSION=99)

foreach(target ZToolsMacNativeAbiFixture ZToolsMacNativeUnsupportedAbiFixture)
  target_include_directories(${target} PRIVATE include)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8 /sdl)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
  endif()
endforeach()

add_executable(ZToolsMacNativeAbiTest fixture/abi_test.cpp)
target_include_directories(ZToolsMacNativeAbiTest PRIVATE include)
if(MSVC)
  target_compile_options(ZToolsMacNativeAbiTest PRIVATE /W4 /WX /utf-8 /sdl)
else()
  target_compile_options(ZToolsMacNativeAbiTest PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(ZToolsMacNativeAbiTest PRIVATE ${CMAKE_DL_LIBS})
endif()

add_library(ZToolsMacNativeWorkerSession STATIC host/worker_session_v1.cpp)
target_include_directories(ZToolsMacNativeWorkerSession PUBLIC include host)
add_executable(ZToolsMacNativeWorkerSessionTest fixture/worker_session_test.cpp)
target_link_libraries(ZToolsMacNativeWorkerSessionTest PRIVATE ZToolsMacNativeWorkerSession)
add_executable(ZToolsMacNativeWorkerKeyTest fixture/worker_key_test.cpp)
target_include_directories(ZToolsMacNativeWorkerKeyTest PRIVATE host)
add_executable(ZToolsMacNativeTestWorker host/test_worker_process.cpp)
target_link_libraries(ZToolsMacNativeTestWorker PRIVATE ZToolsMacNativeWorkerSession)
if(MSVC)
  target_compile_options(ZToolsMacNativeWorkerSession PRIVATE /W4 /WX /utf-8 /sdl)
  target_compile_options(ZToolsMacNativeWorkerSessionTest PRIVATE /W4 /WX /utf-8 /sdl)
  target_compile_options(ZToolsMacNativeWorkerKeyTest PRIVATE /W4 /WX /utf-8 /sdl)
  target_compile_options(ZToolsMacNativeTestWorker PRIVATE /W4 /WX /utf-8 /sdl)
else()
  target_compile_options(ZToolsMacNativeWorkerSession PRIVATE -Wall -Wextra -Werror)
  target_compile_options(ZToolsMacNativeWorkerSessionTest PRIVATE -Wall -Wextra -Werror)
  target_compile_options(ZToolsMacNativeWorkerKeyTest PRIVATE -Wall -Wextra -Werror)
  target_compile_options(ZToolsMacNativeTestWorker PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(ZToolsMacNativeTestWorker PRIVATE ${CMAKE_DL_LIBS})
endif()
